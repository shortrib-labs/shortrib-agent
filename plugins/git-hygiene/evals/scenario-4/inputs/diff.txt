+"""User profile search functionality."""
+
+from typing import Optional
+from .models import UserProfile
+from .database import db
+
+
+def search_profiles(
+    query: str,
+    limit: int = 20,
+    offset: int = 0,
+    active_only: bool = True,
+) -> list[UserProfile]:
+    """Search user profiles by name, email, or username.
+
+    Args:
+        query: Search string matched against name, email, and username fields.
+        limit: Maximum number of results to return (default 20).
+        offset: Number of results to skip for pagination (default 0).
+        active_only: When True, excludes deactivated accounts (default True).
+
+    Returns:
+        List of UserProfile objects matching the search query.
+    """
+    filters = []
+
+    if active_only:
+        filters.append(UserProfile.is_active == True)
+
+    results = (
+        db.session.query(UserProfile)
+        .filter(*filters)
+        .filter(
+            (UserProfile.name.ilike(f"%{query}%"))
+            | (UserProfile.email.ilike(f"%{query}%"))
+            | (UserProfile.username.ilike(f"%{query}%"))
+        )
+        .order_by(UserProfile.name)
+        .limit(limit)
+        .offset(offset)
+        .all()
+    )
+    return results
+
+
+def search_profiles_count(query: str, active_only: bool = True) -> int:
+    """Return total count of profiles matching a search query (for pagination)."""
+    filters = []
+    if active_only:
+        filters.append(UserProfile.is_active == True)
+    return (
+        db.session.query(UserProfile)
+        .filter(*filters)
+        .filter(
+            (UserProfile.name.ilike(f"%{query}%"))
+            | (UserProfile.email.ilike(f"%{query}%"))
+            | (UserProfile.username.ilike(f"%{query}%"))
+        )
+        .count()
+    )
